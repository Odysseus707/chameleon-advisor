import chi
from chi import lease, container
chi.use_site("CHI@Edge")
chi.set("project_name", "CHI-000000")

# Create a lease
res = []
lease_obj = lease.Lease(
    name="demo-lease",
    resources=res,
    start_date=lease.lease_duration(days=0, hours=10)[0],
    end_date=lease.lease_duration(days=0, hours=10)[1]
)
lease_obj.create()

# Wait for the lease to become active
lease_obj.wait_for_active()

# Create a container
container_obj = container.Container(
    name="demo-app",
    image="python:3.9-slim",
    reservation_id=lease_obj.get_device_reservation(),
    exposed_ports=[22],
    platform_version=2
)
container_obj.create()

However, the above code does not perfectly match the original script because the original script uses lease.add_device_reservation to add a device reservation to the list of resources, which is not directly equivalent in the object-oriented API.

Here's a more accurate version:

import chi
from chi import lease, container
chi.use_site("CHI@Edge")
chi.set("project_name", "CHI-000000")

# Create a device reservation
res = []
lease.add_device_reservation(res, machine_name="raspberrypi4-64", count=1)

# Create a lease
lease_obj = lease.Lease(
    name="demo-lease",
    resources=res,
    start_date=lease.lease_duration(days=0, hours=10)[0],
    end_date=lease.lease_duration(days=0, hours=10)[1]
)
lease_obj.create()

# Wait for the lease to become active
lease_obj.wait_for_active()

# Create a container
container_obj = container.Container(
    name="demo-app",
    image="python:3.9-slim",
    reservation_id=lease_obj.get_device_reservation(),
    exposed_ports=[22],
    platform_version=2
)
container_obj.create()

This code maintains the same functionality as the original script but uses the object-oriented API of python-chi. Note that the lease.add_device_reservation function is still used to add a device reservation to the list of resources, as this functionality is not directly replaced in the object-oriented API.
