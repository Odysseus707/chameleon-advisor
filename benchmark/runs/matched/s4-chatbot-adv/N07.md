To reserve two Raspberry Pi 4s in one CHI@Edge lease and start an SSH-enabled container on each of them, follow these steps:

1. **Select Chameleon Site**:
   ```python
   import chi
   chi.use_site("CHI@Edge")
   ```

2. **Set Project Name**:
   ```python
   chi.set("project_name", "Chameleon")
   ```

3. **Create a Lease for Two Raspberry Pi 4 Devices**:
   ```python
   from chi import lease

   machine_name = "raspberrypi4-64"
   start, end = lease.lease_duration(days=1)
   lease_name = f"ssh-{machine_name}-{start}"

   reservations = []
   lease.add_device_reservation(reservations, count=2, machine_name=machine_name)

   container_lease = lease.create_lease(lease_name, reservations)
   lease_id = container_lease["id"]

   print(f"created lease with name {lease_name} and uuid {lease_id}, waiting for it to start. This can take up to 60s.")
   lease.wait_for_active(lease_id)
   print("Done!")
   ```

4. **Start SSH-Enabled Containers on Each Device**:
   ```python
   from chi import container

   # Function to create and configure a container
   def create_ssh_container(container_name, reservation_id):
       try:
           my_container = container.create_container(
               container_name,
               image="ghcr.io/chameleoncloud/edge_ssh_image:latest",
               workdir="/home",
               exposed_ports=[22],
               reservation_id=reservation_id,
               platform_version=2,
           )
           print(f"Successfully created container: {container_name}!")
       except RuntimeError as ex:
           print(ex)
           print(f"please stop and/or delete {container_name} and try again")
       
       # Upload your public key to the authorized keys within the container
       chi.container.upload(my_container.uuid, "./edge_user_public_key", "/root/.ssh/")
       cmd = "/bin/bash -c \"chmod 600 /root/.ssh/authorized_keys\""
       chi.container.execute(my_container.uuid, cmd)
       cmd = "/bin/bash -c \"cat /root/.ssh/edge_user_public_key >> /root/.ssh/authorized_keys\""
       chi.container.execute(my_container.uuid, cmd)

   # Create containers for each reservation
   reservations_info = lease.get_device_reservation(lease_id)
   for i, reservation in enumerate(reservations_info):
       container_name = f"tutorial-{machine_name}-ssh{i+1}".replace("_","-")
       create_ssh_container(container_name, reservation["id"])
   ```

5. **SSH into Each Container**:
   ```python
   # Function to get the IP address and print SSH command
   def ssh_into_container(my_container):
       ip_address = chi.container.associate_floating_ip(my_container.uuid)
       print("use the following command to ssh into the container: ssh root@" + ip_address)

   for i, reservation in enumerate(reservations_info):
       my_container = container.get_container_by_name(f"tutorial-{machine_name}-ssh{i+1}".replace("_","-"))
       ssh_into_container(my_container)
   ```

6. **Destroy Containers After Use**:
   ```python
   # Function to destroy a container
   def destroy_container(container_uuid):
       chi.container.destroy_container(container_uuid)

   for i, reservation in enumerate(reservations_info):
       my_container = container.get_container_by_name(f"tutorial-{machine_name}-ssh{i+1}".replace("_","-"))
       destroy_container(my_container.uuid)
   ```

This process will reserve two Raspberry Pi 4 devices and start an SSH-enabled container on each device. You can then SSH into the containers using the provided IP addresses.
