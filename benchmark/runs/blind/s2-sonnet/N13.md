pythonimport chi
import chi.lease
import chi.container
from datetime import datetime, timedelta

# --- Configuration ---
chi.set("project_name", "CHI-XXXXXX")   # <-- set your project
chi.use_site("CHI@Edge")

lease_name = "pi4-10hr-lease"
container_name = "pi4-py39-ssh"
machine_name = "raspberrypi4-64"        # Pi 4 device type on CHI@Edge

# --- 1. Reserve one Raspberry Pi 4 for 10 hours starting now ---
start_date, end_date = chi.lease.lease_duration(hours=10)

device_reservations = [
    {
        "resource_type": "device",
        "resource_properties": '["=", "machine_name", "%s"]' % machine_name,
        "min": 1,
        "max": 1,
    }
]

reservation_list = chi.lease.add_device_reservation(
    [], count=1, machine_name=machine_name
)

lease = chi.lease.create_lease(
    lease_name,
    reservations=reservation_list,
    start_date=start_date,
    end_date=end_date,
)

lease_id = lease["id"]
chi.lease.wait_for_active(lease_id)

device_reservation_id = chi.lease.get_device_reservation(lease_id)

# --- 2. Launch python:3.9-slim on the reserved Pi 4 ---
container = chi.container.create_container(
    container_name,
    image="python:3.9-slim",
    reservation_id=device_reservation_id,
    exposed_ports=[22],
    command=[
        "sh", "-c",
        "apt-get update && apt-get install -y openssh-server && "
        "mkdir -p /run/sshd && "
        "echo 'root:chameleon' | chpasswd && "
        "sed -i 's/#PermitRootLogin.*/PermitRootLogin yes/' /etc/ssh/sshd_config && "
        "/usr/sbin/sshd -D"
    ],
)

chi.container.wait_for_active(container.id)

# --- 3. Associate a floating IP so SSH is reachable from outside ---
ip = chi.container.associate_floating_ip(container.id)

print("Lease ID:", lease_id)
print("Container ID:", container.id)
print("SSH reachable at:", ip, "port 22")
Notes:

Set chi.set("project_name", ...) to your actual project/allocation before running.
machine_name="raspberrypi4-64" is the CHI@Edge device type string for Pi 4 nodes — confirm against current site device listings (chi.container.list_machines() or chi.lease.list_device_types()) in case naming has changed at your site.
The container command installs and starts sshd at boot since the base python:3.9-slim image doesn't ship an SSH server; adjust the root password / consider injecting an SSH key via environment variables or a custom Dockerfile-based image if you want key-based auth instead of password auth.
associate_floating_ip is what makes the container's port 22 reachable from outside the CHI@Edge private network — you may need to open port 22 in the container's security group if your site enforces one separately.
