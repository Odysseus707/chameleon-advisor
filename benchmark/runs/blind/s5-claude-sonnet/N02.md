```python
import base64
import chi
from chi import lease, container

chi.use_site("CHI@Edge")

reservations = []
lease.add_device_reservation(
    reservations, machine_name="raspberrypi4-64", count=1, device_profiles=["pi_camera"]
)
start_date, end_date = lease.lease_duration(hours=12)
l = lease.create_lease("camera-still-lease", reservations, start_date=start_date, end_date=end_date)
lease.wait_for_active(l["id"])
reservation_id = lease.get_device_reservation(l["id"])

c = container.create_container(
    "camera-still",
    image="your-camera-image",  # must have rpicam-apps / libcamera-apps installed
    reservation_id=reservation_id,
    device_profiles=["pi_camera"],
    exposed_ports=[22],
)
container.wait_for_active(c["uuid"])

container.execute(c["uuid"], "rpicam-still -o /tmp/still.jpg --width 640 --height 480")

result = container.execute(c["uuid"], "base64 /tmp/still.jpg")
with open("still.jpg", "wb") as f:
    f.write(base64.b64decode(result["output"]))
```

`lease_duration(hours=12)` gets you the 12-hour window; `device_profiles=["pi_camera"]` on both the reservation and the container is what actually grants access to `/dev/video*` inside the container. Use `libcamera-still` instead of `rpicam-still` if the image is Bullseye-based rather than Bookworm — the tool was renamed, not re-architected, so the flags are the same.
