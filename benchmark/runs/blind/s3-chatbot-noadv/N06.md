```python
from chi import Lease, Container

# Set up the environment for Chameleon Cloud
chi.use_site("CHI@Edge")
chi.set("project_name", "CHI-000000")

# Create a lease object and add device reservation
l = Lease(name="demo-lease")
device_reservation = l.add_device_reservation(machine_name="raspberrypi4-64", count=1)
start_date, end_date = l.lease_duration(days=0, hours=10)

# Submit the lease request
l.submit(start_date=start_date, end_date=end_date)

# Wait for the lease to become active
l.wait_for_active()

# Create a container using the reservation from the lease
my_container = Container(
    name="demo-app",
    image="python:3.9-slim",
    reservation_id=device_reservation["id"],
    exposed_ports=[22],
    platform_version=2,
)

# Optionally, you can submit the container to be created (if not done automatically)
my_container.submit()
```
